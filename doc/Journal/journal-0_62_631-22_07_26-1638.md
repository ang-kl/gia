    ```js
    let _codeToName = null;
    function codeToName(code) {
      if (!_codeToName) {
        _codeToName = {};
        for (const st of loadStations())
          for (const c of (st.codes || []))
            _codeToName[String(c).toUpperCase()] = st.name;
      }
      return _codeToName[String(code || '').toUpperCase()] || null;
    }
    function resolveStationName(raw) {
      const s = String(raw || '').trim();
      if (/^[A-Z]{1,3}\d+[A-Z]?$/i.test(s)) { const nm = codeToName(s); if (nm) return nm; }
      return normaliseStationName(s);
    }
    ```

    **"Keppel · 1"**, **"Cantonment · 1"**, **"Prince Edward Road · 1"** instead of **"Ne18 · 1"**.
